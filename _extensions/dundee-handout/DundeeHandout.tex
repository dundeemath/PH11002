\makeatletter
\renewcommand{\maketitle}{%
  \begingroup\setlength{\parindent}{0pt}%
  \begin{flushleft}
    {\sffamily\huge\textbf{\MakeUppercase{\@title}}}%
    % subtitle (print only if defined and non-empty)
    \ifdefined\@subtitle
      \ifx\@subtitle\@empty
      \else
        \vspace{0.9ex}\par{\Large \@subtitle}%
      \fi
    \fi
    % author (print only if non-empty)
    \ifdefined\@author
      \ifx\@author\@empty
      \else
        \vspace{0.9ex}\par{\@author}%
      \fi
    \fi
  \end{flushleft}%
  \endgroup
}
\makeatother
